def handle_scan_screenshot():
# Opens a system file dialog to pick an image file
filepath = filedialog.askopenfilename(
title="Select Screenshot to Scan",
filetypes=[("Image Files", ".png;.jpg;.jpeg;.bmp")]
)
if filepath:
try:
results = scan_screenshot(filepath)
if results:
# Format results into a readable list
output = "\n".join(results)
messagebox.showwarning("Sensitive Data Found!", f"Warning: The following data was detected:\n\n{output}")
else:
messagebox.showinfo("Scan Clean", "No sensitive text patterns (emails or phone numbers) detected.")
except Exception as e:
messagebox.showerror("Error", f"Failed to process screenshot:\n{e}")

def handle_scan_file():
# Opens a system file dialog to pick any file
filepath = filedialog.askopenfilename(title="Select File for Hash Scanning")
if filepath:
try:
results = scan_file(filepath)
messagebox.showinfo("File Scan Complete", f"File: {filepath.split('/')[-1]}\n\nSHA256 Hash:\n{results['sha256']}")
except Exception as e:
messagebox.showerror("Error", f"Failed to read file:\n{e}")

import cv2
import easyocr
import re
import numpy as np

Initialize the EasyOCR reader (English language)

reader = easyocr.Reader(['en'], gpu=False)

Target patterns to secure

PATTERNS = {
'Email': r'[\w.+-]+@[\w.-]+.\w+',
'Phone': r'\b[6-9]\d{9}\b'
}

def scan_screenshot(path):
img = cv2.imread(path)
if img is None:
raise FileNotFoundError("Could not open or read the image file.")

res = reader.readtext(path)  
found = []  
  
for b, t, c in res:  
    for k, p in PATTERNS.items():  
        if re.search(p, t):  
            found.append(f"{k}: {t.strip()}")  
return found

import hashlib

def scan_file(path):
# Standard 64KB chunk size to read large files safely without crashing RAM
sha256_hash = hashlib.sha256()
with open(path, "rb") as f:
for byte_block in iter(lambda: f.read(65536), b""):
sha256_hash.update(byte_block)

return {'sha256': sha256_hash.hexdigest()}

def scan_sms(msg):
m = msg.lower()
# Scans for urgency and credential theft keywords
if 'otp' in m or 'verify' in m or 'urgent' in m or 'blocked' in m:
return 'Potential scam detected'
return 'Message appears safe'
